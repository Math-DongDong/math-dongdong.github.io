// ═══════════════════════════════════════════════════════════════════════════
//  MathDongDong — Cloud Firestore 보안 규칙 (v4.2 + 커스텀 설문 응답 쓰기 수정판)
//  records: 체크포인트(isCheckpoint=true)의 최종 확정 갱신을 1회 허용
// ═══════════════════════════════════════════════════════════════════════════

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // ─────────────────────────────────────────────────────────────────────
    //  공통 함수
    // ─────────────────────────────────────────────────────────────────────

    function isAdmin() {
      return request.auth != null
          && request.auth.uid == '81v0uN5YhFOFXtxn9xisGh5CTka2';
    }

    function teacherPath() {
      return /databases/$(database)/documents/teachers/$(request.auth.uid);
    }

    function isTeacher() {
      return isAdmin()
          || (request.auth != null
              && exists(teacherPath())
              && get(teacherPath()).data.status == 'approved');
    }

    function me() {
      return get(teacherPath()).data;
    }

    function isGameCol(name) {
      return name != 'studentsreview_records' && name != 'custom_forms' && name.matches('.*_records');
    }

    function validGameRecord() {
      return request.resource.data.nickname is string
          && request.resource.data.nickname.size() > 0
          && request.resource.data.nickname.size() <= 40
          && request.resource.data.score is number
          && request.resource.data.score >= 0
          && request.resource.data.keys().size() <= 25
          && !request.resource.data.keys().hasAny(['studentId', 'school', 'studentKey']);
    }

    function isRoomCode(code) {
      return code.matches('^[A-Z0-9]{4}$');
    }

    function ownsStudent(data) {
      return isAdmin()
          || (isTeacher()
              && data.school == me().school.trim()
              && me().name.trim() in data.teachers);
    }

    function studentPath(key) {
      return /databases/$(database)/documents/student_auth/$(key);
    }

    function ownsStudentKey(key) {
      return isAdmin()
          || (exists(studentPath(key)) && ownsStudent(get(studentPath(key)).data));
    }

    function roomPath(gameCol, roomCode) {
      return /databases/$(database)/documents/$(gameCol)/$(roomCode);
    }

    function ownsRoom(gameCol, roomCode) {
      return isAdmin()
          || (isTeacher()
              && (!exists(roomPath(gameCol, roomCode))
                  || get(roomPath(gameCol, roomCode)).data.get('createdBy', '') in ['', request.auth.uid]));
    }

    function validStudentAuth() {
      return request.resource.data.keys().hasOnly([
               'school', 'studentId', 'teachers', 'devices',
               'presence', 'lastActive', 'pinSetAt',
               'createdAt', 'updatedAt'
             ])
          && request.resource.data.school is string
          && request.resource.data.school.size() <= 60
          && request.resource.data.studentId is string
          && request.resource.data.studentId.size() <= 12
          && request.resource.data.get('devices', []).size() <= 3
          && request.resource.data.get('teachers', []).size() <= 10
          && request.resource.data.get('presence', 'offline') in ['online', 'away', 'pending', 'offline'];
    }


    // ─────────────────────────────────────────────────────────────────────
    //  teacher_directory/{uid} — 공개 교사 목록
    // ─────────────────────────────────────────────────────────────────────
    match /teacher_directory/{uid} {
      allow get, list: if true;
      allow create, update: if request.resource.data.keys().hasOnly(['school', 'name'])
                            && request.resource.data.school is string
                            && request.resource.data.name is string
                            && request.resource.data.school.size() <= 60
                            && request.resource.data.name.size() <= 40
                            && (isAdmin()
                                || (isTeacher()
                                    && request.auth.uid == uid
                                    && request.resource.data.school.trim() == me().school.trim()
                                    && request.resource.data.name.trim() == me().name.trim()));
      allow delete: if isAdmin() || (request.auth != null && request.auth.uid == uid);
    }


    // ─────────────────────────────────────────────────────────────────────
    //  teachers/{uid} — 교사 계정
    // ─────────────────────────────────────────────────────────────────────
    match /teachers/{uid} {
      allow get: if request.auth != null && (isAdmin() || request.auth.uid == uid);
      allow list: if isAdmin();
      allow create: if request.auth != null
                    && request.auth.uid == uid
                    && request.resource.data.status == 'pending'
                    && request.resource.data.email == request.auth.token.email
                    && request.resource.data.name is string
                    && request.resource.data.name.size() <= 40
                    && request.resource.data.school is string
                    && request.resource.data.school.size() <= 60;
      allow update: if isAdmin()
                    || (request.auth != null
                        && request.auth.uid == uid
                        && request.resource.data.status == resource.data.status
                        && request.resource.data.email == resource.data.email
                        && request.resource.data.name == resource.data.name);
      allow delete: if isAdmin()
                    || (request.auth != null && request.auth.uid == uid);
    }


    // ─────────────────────────────────────────────────────────────────────
    //  student_auth/{학교_학번} — 학생 인증 정보
    // ─────────────────────────────────────────────────────────────────────
    match /student_auth/{key} {
      allow get: if true;
      allow list: if ownsStudent(resource.data);
      allow create: if validStudentAuth()
                    && request.resource.data.studentId.matches('^[0-9]{4,5}$');
      allow update: if validStudentAuth()
                    && request.resource.data.school == resource.data.get('school', request.resource.data.school)
                    && request.resource.data.studentId == resource.data.get('studentId', request.resource.data.studentId);
      allow delete: if ownsStudent(resource.data);

      match /private/auth {
        allow get, list: if false;
        allow create: if ownsStudentKey(key)
                      || (request.resource.data.pinHash is string
                          && request.resource.data.pinHash.size() == 64);
        allow update: if ownsStudentKey(key)
                      || (resource.data.get('pinHash', null) == null
                          && request.resource.data.pinHash is string
                          && request.resource.data.pinHash.size() == 64);
        allow delete: if ownsStudentKey(key);
      }

      match /gate/{deviceFp} {
        allow read: if false;
        allow create, update: if request.resource.data.proof is string
                              && request.resource.data.proof ==
                                 get(/databases/$(database)/documents/student_auth/$(key)/private/auth).data.pinHash;
        allow delete: if ownsStudentKey(key);
      }
    }


    // ─────────────────────────────────────────────────────────────────────
    //  custom_forms/{방} — 동적 커스텀 폼(설문조사) 시스템 규칙
    // ─────────────────────────────────────────────────────────────────────
    match /custom_forms/{roomCode} {
      // 학생 및 교사는 룸 코드(상태, 모드, 문항 양식) 조회 가능
      allow get: if true;
      // 교사는 본인이 생성한 방 목록 및 관리자 전체 목록 조회 가능
      allow list: if isTeacher();

      // 방 생성: 승인된 교사/관리자만 가능
      allow create: if isTeacher()
                    && isRoomCode(roomCode)
                    && request.resource.data.createdBy == request.auth.uid;

      // 방 정보 수정:
      // 1) 방을 만든 교사 또는 관리자
      // 2) 학생 응답 제출 시 responseCount 및 students 명단 업데이트 허용 (★권한 거부 해결)
      allow update: if isAdmin()
                    || (isTeacher() && resource.data.get('createdBy', '') in ['', request.auth.uid])
                    || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['responseCount', 'students']));

      // 방 삭제: 방을 만든 교사 또는 관리자만 가능
      allow delete: if ownsRoom('custom_forms', roomCode);

      // ── students/{nickname} — 참가 학생 및 게스트 승인 정보 ────────────
      match /students/{nickname} {
        // 학생 정보는 해당 설문 방 소유 교사와 관리자만 조회
        allow get, list: if ownsRoom('custom_forms', roomCode);

        // 학생 입장 과정에서 참가 정보 생성·갱신
        allow create, update: if request.resource.data.nickname is string
                              && request.resource.data.nickname.size() <= 40
                              && request.resource.data.keys().size() <= 25;

        // 방 소유 교사와 관리자만 참가 정보를 삭제
        allow delete: if ownsRoom('custom_forms', roomCode);
      }

      // ── guests/{nickname} — 게스트 승인 상태 ────────────────────────────
      match /guests/{nickname} {
        // 게스트 학생은 자신의 승인 상태를 확인
        allow get: if true;
        // 교사는 승인 대기 목록을 조회
        allow list: if ownsRoom('custom_forms', roomCode);

        // 게스트는 pending 상태만 요청 가능. 방 소유 교사는 승인/거절 가능
        allow create, update: if ownsRoom('custom_forms', roomCode)
                              || (request.resource.data.keys().hasOnly(['status'])
                                  && request.resource.data.status == 'pending');

        // 교사는 언제든 삭제 가능. 게스트는 대기 중 요청만 취소 가능
        allow delete: if ownsRoom('custom_forms', roomCode)
                      || resource.data.get('status', '') == 'pending';
      }

      // ── responses/{studentId} — 학생 응답 결과 ─────────────────────────
      match /responses/{studentId} {
        // 응답 조회: 학생 본인 조회 및 교사 응답 목록 조회/엑셀 다운로드 허용
        allow get, list: if true;

        // 응답 제출 및 수정: 학생(빠른 입장/인증 입장 공통) 작성 허용
        allow create, update: if request.resource.data.studentId is string
                              && request.resource.data.studentName is string;

        // 응답 삭제: 방을 만든 교사 또는 관리자만 가능
        allow delete: if ownsRoom('custom_forms', roomCode);
      }
    }


    // ─────────────────────────────────────────────────────────────────────
    //  studentsreview_records/{방} — 소감문 관리 시스템 전용 규칙
    // ─────────────────────────────────────────────────────────────────────
    match /studentsreview_records/{roomCode} {
      allow get: if true;
      allow list: if isTeacher();
      allow create: if isTeacher()
                    && isRoomCode(roomCode)
                    && request.resource.data.createdBy == request.auth.uid;
      allow update, delete: if ownsRoom('studentsreview_records', roomCode);

      match /records/{recordId} {
        allow get, list: if true;
        allow create, update: if ownsRoom('studentsreview_records', roomCode)
                              || (request.resource.data.studentId is string
                                  && request.resource.data.content is string);
        allow delete: if ownsRoom('studentsreview_records', roomCode);
      }

      match /students/{nickname} {
        allow get, list: if ownsRoom('studentsreview_records', roomCode);
        allow create, update: if request.resource.data.nickname is string
                              && request.resource.data.keys().size() <= 25;
        allow delete: if ownsRoom('studentsreview_records', roomCode);
      }

      match /nicknames/{nickname} {
        allow get: if true;
        allow list: if isTeacher();
        allow create: if request.resource.data.keys().size() == 0;
        allow update: if false;
        allow delete: if ownsRoom('studentsreview_records', roomCode);
      }

      match /guests/{nickname} {
        allow get: if true;
        allow list: if isTeacher();
        allow create, update: if ownsRoom('studentsreview_records', roomCode)
                              || (request.resource.data.keys().hasOnly(['status'])
                                  && request.resource.data.status == 'pending');
        allow delete: if true;
      }
    }


    // ─────────────────────────────────────────────────────────────────────
    //  {게임}_records/{방} — 기존 게임 방 및 익명 순위표 규칙
    // ─────────────────────────────────────────────────────────────────────
    match /{gameCol}/{roomCode} {
      allow get: if isGameCol(gameCol);
      allow list: if isGameCol(gameCol) && isTeacher();
      allow create: if isGameCol(gameCol) && isTeacher()
                    && isRoomCode(roomCode)
                    && request.resource.data.createdBy == request.auth.uid
                    && (isAdmin()
                        || request.resource.data.get('roomMode', 'quick') != 'auth'
                        || request.resource.data.get('creatorSchool', '') == me().school.trim());

      allow update: if isGameCol(gameCol)
                    && (isAdmin()
                        || (isTeacher()
                            && resource.data.get('createdBy', '') in ['', request.auth.uid]
                            && request.resource.data.get('creatorSchool', '') == resource.data.get('creatorSchool', '')));

      allow delete: if isGameCol(gameCol)
                    && (isAdmin()
                        || (isTeacher()
                            && resource.data.get('createdBy', '') in ['', request.auth.uid]));

      match /students/{nickname} {
        allow get, list: if isGameCol(gameCol) && ownsRoom(gameCol, roomCode);
        allow create, update: if isGameCol(gameCol)
                              && request.resource.data.nickname is string
                              && request.resource.data.nickname.size() <= 40
                              && request.resource.data.keys().size() <= 25;
        allow delete: if isGameCol(gameCol) && ownsRoom(gameCol, roomCode);
      }

      match /nicknames/{nickname} {
        allow get: if isGameCol(gameCol);
        allow list: if isGameCol(gameCol) && isTeacher();
        allow create: if isGameCol(gameCol)
                      && request.resource.data.keys().size() == 0;
        allow update: if false;
        allow delete: if isGameCol(gameCol) && ownsRoom(gameCol, roomCode);
      }

      match /guests/{nickname} {
        allow get: if isGameCol(gameCol);
        allow list: if isGameCol(gameCol) && isTeacher();
        allow create, update: if isGameCol(gameCol)
                              && (ownsRoom(gameCol, roomCode)
                                  || (request.resource.data.keys().hasOnly(['status'])
                                      && request.resource.data.status == 'pending'));
        allow delete: if isGameCol(gameCol);
      }

      match /records/{recordId} {
        allow get, list: if isGameCol(gameCol);
        allow create: if isGameCol(gameCol)
                      && validGameRecord();

        // 공개 클라이언트는 체크포인트를 최종 기록으로 한 번만 확정할 수 있습니다.
        // 소유 인증이 없는 구조이므로 닉네임을 고정하고, 민감 필드·기록 스키마를 재검증합니다.
        allow update: if isGameCol(gameCol)
                      && resource.data.get('isCheckpoint', false) == true
                      && request.resource.data.keys().hasAll(['isCheckpoint'])
                      && request.resource.data.isCheckpoint == false
                      && request.resource.data.nickname == resource.data.nickname
                      && validGameRecord();

        allow delete: if isGameCol(gameCol) && ownsRoom(gameCol, roomCode);
      }
    }


    // 명시하지 않은 경로는 전부 거부됩니다(기본값).
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
